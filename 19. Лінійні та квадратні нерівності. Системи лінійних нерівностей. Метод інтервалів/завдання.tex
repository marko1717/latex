\begin{samepage}
% НМТ 2026, сесія 23.05, завдання №11
\zadtask{Знайдіть найбільший цілий розв'язок нерівності $(x - 4)(4 + x) \leqslant 8x - 7$. \nmtyear{2026}}
\answerTable{$0$}{$9$}{$-1$}{$10$}{$8$}
% Відповідь: Б
\ifshowsolutions\solution{$(x-4)(x+4)\le 8x-7 \Rightarrow x^2-16\le 8x-7 \Rightarrow x^2-8x-9\le 0$. Дискримінант $D=64+36=100$, корені $x=\dfrac{8\pm 10}{2}$, тобто $-1$ і $9$. Отже $-1\le x\le 9$, найбільший цілий~--- $9$. Відповідь: \textbf{Б}.}\fi
\par\penalty-20
\end{samepage}
